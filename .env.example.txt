############################################################################
#  Rotation                                                                 #
############################################################################
# Seconds a healthy exit carries traffic before rotating. Rotation costs a
# hard cut of every in-flight connection, so unless a fresh exit IP is worth
# that on its own, do not pay it often. The tunnel is watched continuously
# regardless (VPN_WATCHDOG_INTERVAL), and a degraded exit is rotated away by
# the scorer rather than by this timer.
# NOTE: shortening this means also raising BASELINE_ALPHA — they pair.
# ROTATION_INTERVAL=86400

# Seconds to wait for tun to come up before giving up on an exit.
# VPN_CONNECT_TIMEOUT=45

# Glob used to find .ovpn files in /etc/openvpn/configs.
# OVPN_FILE_PATTERN=*.ovpn

# OpenVPN keepalive. The ProtonVPN .ovpn files ship none, so without these a
# silently dead UDP tunnel is invisible to OpenVPN itself — and `persist-tun`
# keeps the interface up with its IP, so it looks alive from outside too.
# This is the only cover for a black-holed-but-alive tunnel; the kill switch
# cannot see that case.
# VPN_PING_INTERVAL=10
# VPN_PING_RESTART=60

# Seconds in-flight requests get to finish before the listener is killed and
# the tunnel torn down. At an 86400 cycle a 30s drain costs 0.03% of uptime.
#
# COUPLED TO docker-compose.yml: stop_grace_period must exceed DRAIN_GRACE
# plus the OpenVPN teardown (~5s). Docker's default stop timeout is 10s, so
# raising this above ~5 without raising stop_grace_period guarantees the
# container is SIGKILLed mid-drain and never shuts down cleanly.
# DRAIN_GRACE=30

############################################################################
#  Health / metrics listener                                                #
############################################################################
# GET /health, /ready, /metrics. Binds 0.0.0.0 inside the container's network
# namespace; what matters is that docker-compose.yml publishes it to 127.0.0.1
# rather than to every host interface.
# HEALTH_HTTP_ENABLED=1
# HEALTH_PORT=8081
# HEALTH_LISTEN_ADDR=0.0.0.0
# Cap on concurrent handler processes; `fork` with no cap is a trivial DoS.
# HEALTH_MAX_CHILDREN=16
# Idle timeout (s) for a connection that never completes a request.
# HEALTH_IO_TIMEOUT=5
# Max header lines read before the handler stops draining.
# HEALTH_MAX_HEADERS=64
# /ready returns 503 if the supervisor has not refreshed its state within this
# many seconds. The supervisor heartbeats every VPN_WATCHDOG_INTERVAL, so this
# catches a wedged supervisor loop — the one failure where everything else
# still looks fine (privoxy up, tunnel up, last state "ready"). Must comfortably
# exceed VPN_WATCHDOG_INTERVAL or you will get spurious 503s.
# HEALTH_MAX_STATE_AGE=60


############################################################################
#  Tunnel watchdog (kill switch)                                            #
############################################################################
# How often the egress path is verified, and therefore how long a leak or a
# dead tunnel can last. Independent of ROTATION_INTERVAL — keep it small.
# Catches: OpenVPN dying, and routes reverting off the tunnel.
# Does NOT catch: a black-holed tunnel with OpenVPN still alive — that is what
# VPN_PING_RESTART and the error-rate trip are for.
# VPN_WATCHDOG_INTERVAL=5
# Address used for the `ip route get` egress check. Must be a public IP that
# is not inside any pushed VPN subnet.
# VPN_ROUTE_PROBE_IP=1.1.1.1


############################################################################
#  Endpoint latency scoring and degradation                                 #
#  Thresholds are RATIOS, not milliseconds — the fleet defines "slow".       #
############################################################################
# Master switch. 0 = plain round-robin rotation with no ejection logic. The
# kill switch, drain and clean shutdown are NOT gated by this and stay active.
# ENDPOINT_SCORING_ENABLED=1

# Where the latency history and degradation state live. Mount a volume, or
# every restart relearns from zero — at a long rotation interval that means
# never accumulating a usable fleet baseline at all.
# STATS_DIR=/var/lib/proxy

# --- Measurement ---
# Kernel RTT of live proxied sockets. INFORMATIONAL ONLY — exported as
# proxy_passive_rtt_ms, never scored. It measures a single transport round
# trip (~5ms) while the probe measures a full proxied request (~33ms);
# averaging both into one EWMA made an endpoint's score depend on whether
# client traffic happened to be flowing rather than on the endpoint.
# PASSIVE_RTT_ENABLED=1
# Established sockets required before a passive observation is reported.
# PASSIVE_MIN_SOCKETS=2
# Latency sampling cadence, decoupled from the watchdog. Sets both how much
# wall-clock the statistics span (EWMA ~9 samples, baseline ~400) and how
# often the scoring probe fires. At 60s that is 1440 probes/day, split
# round-robin across PROBE_TARGETS.
# SAMPLE_INTERVAL=60
# Probe targets. Plain HTTP, small bodies: TTFB should reflect the path, not
# TLS setup or payload size. Space-separated, rotated round-robin.
# PROBE_TARGETS=http://ipv4.icanhazip.com/ http://ifconfig.me/ip
# A probe with no first byte inside this many seconds counts as a failure.
# PROBE_TIMEOUT=8

# --- Rolling statistics ---
# Fast EWMA weight (~9-sample memory): catches a degrading exit within minutes.
# EWMA_ALPHA=0.2
# Slow baseline weight (~400 samples = ~6.6h at SAMPLE_INTERVAL=60), i.e. the
# endpoint's own normal. Must span longer than the diurnal cycle or a genuine
# evening peak reads as degradation. Pairs with ROTATION_INTERVAL.
# BASELINE_ALPHA=0.005
# Failure-indicator weight. At 0.10 it takes three consecutive failures to
# cross ERR_TRIP_RATE; raising it lets a single transient error eject an
# endpoint for the full cooldown.
# ERR_ALPHA=0.1

# --- What counts as slow ---
# Trip when the endpoint EWMA exceeds this multiple of the FLEET MEDIAN.
# 2.5 tolerates real geographic spread (a 40ms EU exit and a 180ms APAC exit
# differ 4.5x with nothing wrong) while still catching degradation, because a
# trip ALSO requires the endpoint to be bad against its own frozen baseline.
# Set to 99 to keep scoring and error-trips running with latency ejection
# switched off explicitly — useful while a fleet baseline is still forming.
# SLOW_TRIP_RATIO=2.5
# Re-admit only below this multiple. The 1.5-2.5 gap IS the hysteresis band;
# without it an endpoint at the threshold oscillates every sample.
# SLOW_RECOVER_RATIO=1.5
# Sigma above the endpoint's own baseline also required to trip. This is what
# separates "far away" from "degraded".
# SLOW_SIGMA=3.0
# Never assume a spread tighter than this fraction of the baseline; a very
# stable endpoint would otherwise trip on a single blip.
# SIGMA_FLOOR_FRAC=0.15
# ...nor wider than this, or a noisy endpoint becomes untrippable no matter
# how far it drifts.
# SIGMA_CEIL_FRAC=0.5
# Ratio guard, not a slowness threshold: below this absolute value ratios are
# meaningless, so a 60ms exit is never ejected for being 3x a 20ms median.
# SLOW_FLOOR_MS=150
# Error-rate trip/recover. Absolute, because a 20% failure rate needs no
# relative comparison. Recovery at 0.05 means ~19 clean samples to clear.
# ERR_TRIP_RATE=0.20
# ERR_RECOVER_RATE=0.05

# --- Hysteresis ---
# Samples required before an endpoint can trip at all (20 min at 60s sampling).
# MIN_SAMPLES=20
# Relaxed gate during a half-open trial so a short trial can still conclude.
# HALF_OPEN_MIN_SAMPLES=8
# Base cooldown before an ejected endpoint gets a half-open trial. At a long
# rotation interval this rarely binds — natural rotation is slower than it.
# EJECT_COOLDOWN=900
# Cooldown doubles per consecutive trip, capped here (4h) so a fixed server
# still returns the same day. This exponential term is the real anti-flap.
# EJECT_COOLDOWN_MAX=14400
# Length of a half-open trial cycle: a still-broken exit costs 60s of degraded
# service instead of a full interval.
# HALF_OPEN_INTERVAL=60
# Clean cycles before the backoff exponent resets to zero.
# TRIP_DECAY_CYCLES=3
# Circuit breaker: if more than this fraction of the fleet is ejected, freeze
# new ejections. Half the fleet failing at once is far less likely than a
# problem on this host, and unchecked ejection is how these systems self-DoS.
# MAX_EJECT_FRACTION=0.5

############################################################################
#  Rotation                                                                 #
############################################################################
# Seconds a healthy exit carries traffic before rotating.
#
# Long intervals (86400 = daily) are a reasonable choice: rotation costs a
# hard cut of every in-flight connection, and unless you specifically need a
# fresh exit IP there is little reason to pay it often. If you do run a long
# interval, note that the rotation loop is NOT your tunnel health check —
# VPN_WATCHDOG_INTERVAL is, and it runs regardless of cycle length. Before
# that watchdog existed, a long interval meant a dead tunnel went unnoticed
# for up to a full cycle.
#
# With scoring enabled you can also treat rotation as failure-driven rather
# than scheduled: set this very high and let degradation detection do the
# rotating. See SLOW_TRIP_RATIO below.
#
# NOTE: three values below must move together with this one. This file ships
# the LONG-CYCLE set (86400 / BASELINE_ALPHA=0.005 / DRAIN_GRACE=30). If you
# drop back to a short interval, revert those two to the code defaults noted
# beside them, or the statistics will be tuned for the wrong timescale.
ROTATION_INTERVAL=86400
# Seconds to wait for tun to come up before giving up on an exit.
VPN_CONNECT_TIMEOUT=45
# Glob used to find .ovpn files in /etc/openvpn/configs.
OVPN_FILE_PATTERN=*.ovpn
# OpenVPN keepalive. The ProtonVPN .ovpn files ship none, so without these a
# silently dead UDP tunnel is invisible until the rotation interval elapses.
VPN_PING_INTERVAL=10
VPN_PING_RESTART=60

############################################################################
#  Endpoint latency scoring and degradation                                 #
#  Every knob below has a default in app/endpoint_stats.sh; uncomment to     #
#  override. Ratios, not milliseconds, so the fleet defines what "slow" is.  #
############################################################################

# Master switch. 0 = plain round-robin rotation, the pre-scoring behaviour.
# ENDPOINT_SCORING_ENABLED=1

# --- Where state lives. Mount a volume or every restart relearns from zero. ---
# STATS_DIR=/var/lib/proxy

# --- Measurement ---
# Tunnel kill-switch cadence, and the granularity at which a dead tunnel is
# detected. Independent of ROTATION_INTERVAL — keep it small.
# VPN_WATCHDOG_INTERVAL=5
# Latency sampling cadence, decoupled from the watchdog. This sets how much
# wall-clock the statistics span: EWMA covers ~9 samples, the baseline ~50.
# At 60s that is a ~9min EWMA against a ~100min baseline.
#   ROTATION_INTERVAL <= 900   -> 15 is fine
#   ROTATION_INTERVAL >= 86400 -> keep 60, and see BASELINE_ALPHA below
# SAMPLE_INTERVAL=60
# Use kernel RTT of real proxied sockets as the primary signal (passive).
# PASSIVE_RTT_ENABLED=1
# Minimum established sockets before a passive sample is trusted; below this
# the median is too noisy and we fall back to the in-path probe.
# PASSIVE_MIN_SOCKETS=2
# Seconds between in-path TTFB probes, used only when live traffic is thin.
# PROBE_INTERVAL=15
# Probe targets. Plain HTTP, small bodies: TTFB should reflect the path, not
# TLS setup or payload size. Space-separated, rotated round-robin.
# PROBE_TARGETS=http://ipv4.icanhazip.com/ http://ifconfig.me/ip
# A probe with no first byte inside this many seconds counts as a failure.
# PROBE_TIMEOUT=8

# --- Rolling statistics ---
# Fast EWMA weight (~9-sample memory): catches a degrading exit within a cycle.
# EWMA_ALPHA=0.2
# Slow EWMA weight (~50-sample memory): the endpoint's "own normal". Must be
# much slower than EWMA_ALPHA or a degradation drags the baseline along with it
# and never trips.
#
# Set for the 86400 interval this file ships: 0.005 spans ~400 samples =
# ~6.6h at SAMPLE_INTERVAL=60. A baseline shorter than the diurnal cycle means
# a genuine evening traffic peak is compared against an afternoon baseline and
# reads as degradation — the most likely source of false ejections here.
# Code default is 0.02, which is correct for ROTATION_INTERVAL <= ~900.
BASELINE_ALPHA=0.005
# Failure-indicator EWMA weight. Coupled to ERR_TRIP_RATE below: at 0.10 it
# takes three consecutive failures to cross the trip rate. Raising this makes
# a single transient error enough to eject an endpoint for the full cooldown.
# ERR_ALPHA=0.1

# --- What counts as slow ---
# Trip when the endpoint's EWMA exceeds this multiple of the FLEET MEDIAN.
# 2.5 tolerates genuine geographic spread (a 40ms EU exit vs a 180ms APAC exit
# differ 4.5x with nothing wrong) while still catching real degradation.
# SLOW_TRIP_RATIO=2.5
# Re-admit only below this multiple. The 1.5–2.5 gap IS the hysteresis band;
# without it an endpoint sitting at the threshold oscillates every probe.
# SLOW_RECOVER_RATIO=1.5
# Trip also requires the EWMA to be this many sigma above the endpoint's own
# baseline. This is what separates "far away" from "degraded".
# SLOW_SIGMA=3.0
# Never assume a spread tighter than this fraction of the baseline; a very
# stable endpoint would otherwise trip on a single blip.
# SIGMA_FLOOR_FRAC=0.15
# Ratio guard, not a slowness threshold: below this absolute value ratios are
# meaningless, so a 60ms exit is never ejected for being 3x a 20ms median.
# SLOW_FLOOR_MS=150
# Error-rate trip/recover. Absolute because a 20% failure rate needs no
# relative comparison to be unacceptable. Recovery at 0.05 means ~16 clean
# samples to clear a trip — the hysteresis on the error path.
# ERR_TRIP_RATE=0.20
# ERR_RECOVER_RATE=0.05

# --- Hysteresis ---
# Samples required before an endpoint is eligible to trip at all.
# MIN_SAMPLES=20
# Relaxed gate during a half-open trial, so a short trial can still conclude.
# HALF_OPEN_MIN_SAMPLES=8
# Base cooldown before an ejected endpoint gets a half-open trial (3 cycles).
# EJECT_COOLDOWN=900
# Cooldown doubles per consecutive trip, capped here (4h) so a fixed server
# still returns the same day. This exponential term is the real anti-flap.
# EJECT_COOLDOWN_MAX=14400
# Length of a half-open trial cycle. Short, so a still-broken exit costs 60s
# of degraded service instead of a full 300s interval.
# HALF_OPEN_INTERVAL=60
# Clean cycles before the backoff exponent resets to zero.
# TRIP_DECAY_CYCLES=3
# Circuit breaker: if more than this fraction of the fleet is ejected, freeze
# new ejections. Half of ProtonVPN failing at once is far less likely than a
# problem on this host, and unchecked ejection is how these systems self-DoS.
# MAX_EJECT_FRACTION=0.5

# --- Draining ---
# Seconds in-flight requests get to finish before the listener is killed and
# the tunnel torn down. Rotation used to cut every connection instantly.
# Code default is 10, sized for a 300s cycle where drain is a real fraction of
# uptime. At 86400 a 30s drain costs 0.03% — buy the longer one.
DRAIN_GRACE=30
